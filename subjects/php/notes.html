الزيادة والنقصان

$x = 10; echo ++$x; echo $x;   // 11  11
$x = 10; echo $x++; echo $x;   // 10  11
$x = 10; echo --$x; echo $x;   // 9   9
$x = 10; echo $x--; echo $x;   // 10  9
انتبهفي جدول المحاضرة 2 وصف ++$x و$x++ مكتوب بالعكس. الصحيح: ++$x يزيد ثم يرجع القيمة، و$x++ يرجع القيمة ثم يزيد، وهذا اللي يثبته مثال الشريحة 16.

النماذج Forms و $_POSTتمارين Forms

النموذج يرسل اللي كتبه المستخدم للسيرفر، وPHP تقرأه من مصفوفة $_POST (أو $_GET) باستخدام اسم الحقل name.

  • method="post": البيانات ما تظهر في الرابط. method="get": تظهر في الرابط بعد ?.
  • action="" (أو بدون action): يرسل النموذج لنفس الصفحة.
  • isset($_POST['calc']) تتأكد إن الزر انضغط قبل ما نقرأ القيم، وإلا يطلع تحذير Undefined index أول ما تفتح الصفحة.
  • أزرار radio لها نفس name وقيم value مختلفة، فيرجع واحد فقط. والقائمة <select> ترجع value الخيار المختار.
<form method="post">
  Degree:
  <input type="radio" name="degree" value="phd"> Phd
  <input type="radio" name="degree" value="master"> Master
  Hours: <input type="number" name="hours">
  <input type="submit" name="calc" value="Calc">
</form>

<?php
if (isset($_POST['calc'])) {
    $degree = $_POST['degree'];
    $hours  = $_POST['hours'];
    echo "$degree - $hours";
}
?>
انتبهلو المستخدم ما اختار أي radio، ما ينرسل degree أصلاً. تحقق بـ isset($_POST['degree']) قبل استخدامه. وحلول تمرينَي الفورم في قسم المسائل تحت.

<?php
// 1 من 10 إلى 20 بـ while

while ($i <= 20) { echo $i . "<br>"; $i++; }

// 2 مجموع 1 إلى 9

echo "Sum = $sum <br>";                  // 45

// 3 الزوجية بين 1 والعدد

for ($i = 2; $i <= $n; $i += 2) echo $i . " ";

// 4 الأكبر أس الأصغر
$a = 2; $b = 5;
if ($a > 0 && $b > 0) {
    $big = max($a, $b); $small = min($a, $b);
    $result = 1;
    for ($i = 1; $i <= $small; $i++) $result *= $big;
    echo "$big ^ $small = $result";        // 5 ^ 2 = 25
} else {
    echo "Numbers must be greater than zero";
}

// 5 المضروب
$n = 5; $fact = 1;
for ($i = 1; $i <= $n; $i++) $fact *= $i;
echo "$n! = $fact";                        // 120

// 6 المثلث

    for ($j = 1; $j <= $i; $j++) echo "*";
    echo "<br>";
}
for ($i = 4; $i >= 1; $i--) {
    for ($j = 1; $j <= $i; $j++) echo "*";

}
