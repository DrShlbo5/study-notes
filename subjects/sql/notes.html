  • SQL: لغة قياسية لتخزين ومعالجة واسترجاع البيانات من قواعد البيانات العلائقية.

  • > ANY أكبر من أي قيمة (أي أكبر من الأصغر)، و> ALL أكبر من الكل (أي أكبر من الأكبر).
  • UNION يجمع النتائج بدون تكرار، وINTERSECT المشترك، وEXCEPT/MINUS الموجود في الأول وليس في الثاني.

أساسيات PL/SQL وبنية الكتلةLab 3 · Tut 3

PL/SQL لغة إجرائية من Oracle تجمع أوامر SQL مع خصائص اللغات الإجرائية: المتغيرات، والشروط (IF)، والتكرار (LOOP)، والاستثناءات (Exceptions). البرنامج اسمه كتلة (Block).

DECLARE                    -- اختياري: تعريف المتغيرات
   v_name   VARCHAR2(20) := 'Ali';
   v_salary NUMBER := 5000;
BEGIN                      -- إجباري: التنفيذ
   DBMS_OUTPUT.PUT_LINE('Welcome, ' || v_name);
EXCEPTION                  -- اختياري: معالجة الأخطاء
   WHEN OTHERS THEN DBMS_OUTPUT.PUT_LINE('Error');
END;                       -- إجباري
  • DECLARE اختياري، وBEGIN ... END إجباري، وEXCEPTION اختياري داخل الجسم.
  • كتلة مجهولة (Anonymous): بدون اسم، تنفذ مرة ولا تُخزن ولا تُستدعى. كتلة باسم (Named): تُخزن وتُستدعى مثل Procedure وFunction وPackage.
  • PL/SQL غير حساسة لحالة الأحرف (NOT case-sensitive).
  • مكونات الجمل: الكلمات المحجوزة (BEGIN, IF ...)، والمعرّفات (أسماء يعرّفها المستخدم)، والمحددات (; ( ) ,)، والقيم الثابتة Literals (أرقام ونصوص وتواريخ)، والتعليقات -- و/* */.
  • الطباعة بـ DBMS_OUTPUT.PUT_LINE، والربط بـ ||.
Tut 3 · Task 1تنفيذ DBMS_OUTPUT.PUT_LINE('Hello PLSQL'); لوحدها يعطي خطأ، لأنها جملة PL/SQL ولازم تكون داخل BEGIN ... END;

المتغيرات وأنواع البياناتLab 4 · Tut 4

  • المتغير مكان في الذاكرة لقيمة من نوع محدد، ويُعرّف في قسم DECLARE.
  • معامل الإسناد في PL/SQL هو :=، والمقارنة بـ =.
  • Scalar: قيمة واحدة مثل NUMBER وVARCHAR2 وDATE وBOOLEAN. Composite: بيانات مركبة مثل Records وCollections.
  • التهيئة: إعطاء المتغير قيمة ابتدائية، إما عند التعريف في DECLARE أو داخل BEGIN. المتغير بدون قيمة يكون NULL.
  • DEFAULT: قيمة ابتدائية يحتفظ بها المتغير إذا لم تتغير: v_bonus NUMBER DEFAULT 500;
  • CONSTANT: ثابت يجب إسناد قيمته عند التعريف ولا يمكن تغييرها: c_tax_rate CONSTANT NUMBER := 0.15;
  • SYSDATE تاريخ اليوم، وNULL; جملة لا تفعل شيئاً.

الكتل المتداخلة ونطاق المتغير

DECLARE
   v_message VARCHAR2(20) := 'Outer Block';
BEGIN
   DBMS_OUTPUT.PUT_LINE(v_message);        -- Outer Block
   DECLARE
      v_message VARCHAR2(20) := 'Inner Block';
   BEGIN
      DBMS_OUTPUT.PUT_LINE(v_message);     -- Inner Block
   END;
   DBMS_OUTPUT.PUT_LINE(v_message);        -- Outer Block
END;

متغير الكتلة الخارجية Global يُرى في الداخلية، ومتغير الداخلية Local لا يُرى خارجها. إذا تكرر الاسم، الداخلي يغطي على الخارجي داخل كتلته فقط.

طباعة BOOLEANفي أغلب إصدارات Oracle ما تقدر تطبع BOOLEAN مباشرة بـ PUT_LINE أو تربطها بـ ||. استخدم IF: IF v_flag THEN DBMS_OUTPUT.PUT_LINE('TRUE'); END IF;

الممارسات الجيدة: أسماء واضحة (v_salary بدل x1)، وتعليقات، وتنسيق بإزاحة، وCONSTANT للقيم الثابتة، وتهيئة المتغيرات.

SQL داخل PL/SQLLab 5 · Tut 5

1) EXECUTE IMMEDIATE لأوامر DDL

BEGIN
   EXECUTE IMMEDIATE 'CREATE TABLE departments (dept_id INT PRIMARY KEY, dept_name VARCHAR2(50))';
EXCEPTION
   WHEN OTHERS THEN DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;

أوامر DDL (CREATE, ALTER, DROP, TRUNCATE) لا تُكتب مباشرة داخل الكتلة، لازم EXECUTE IMMEDIATE والأمر كنص بين ' '.

2) SELECT INTO

DECLARE
   v_name   employees.first_name%TYPE;
   v_salary employees.salary%TYPE;
BEGIN
   SELECT first_name, salary INTO v_name, v_salary
   FROM employees WHERE employee_id = 1;
   DBMS_OUTPUT.PUT_LINE('Employee: ' || v_name || ' Salary: ' || v_salary);
END;
  • تجلب البيانات من الجدول وتسندها إلى متغيرات، وترجع صفاً واحداً فقط.
  • لو ما رجع ولا صف: NO_DATA_FOUND. لو رجع أكثر من صف: TOO_MANY_ROWS.

3) %TYPE و %ROWTYPE

  • v_name emp.name%TYPE; المتغير يأخذ نفس نوع العمود في الجدول.
  • emp_rec employees%ROWTYPE; متغير يمثل صفاً كاملاً، وتصل للأعمدة بـ emp_rec.first_name.
DECLARE
   emp_rec employees%ROWTYPE;
BEGIN
   SELECT * INTO emp_rec FROM employees WHERE employee_id = 2;
   DBMS_OUTPUT.PUT_LINE(emp_rec.first_name || ' - ' || emp_rec.salary);
END;

4) DML وعدة صفوف

-- لعرض عدة صفوف نستخدم FOR ... IN (SELECT) LOOP
BEGIN
   FOR rec IN (SELECT first_name, last_name FROM employees) LOOP
      DBMS_OUTPUT.PUT_LINE(rec.first_name || ' ' || rec.last_name);
   END LOOP;
END;

DECLARE
   v_bonus NUMBER := 1000;
BEGIN
   UPDATE employees SET salary = salary + v_bonus WHERE department = 'HR';
   DBMS_OUTPUT.PUT_LINE(SQL%ROWCOUNT || ' employees updated.');   -- عدد الصفوف المتأثرة
   COMMIT;
END;

-- تمرين 3: السعر بعد الخصم
DECLARE
   v_price    NUMBER := 200;
   v_discount NUMBER := 0.10;
   v_final    NUMBER;
BEGIN
   v_final := v_price - (v_price * v_discount);
   DBMS_OUTPUT.PUT_LINE('Final price = ' || v_final);     -- 180
END;

-- تمرين 6 و 8
DECLARE
   c_tax_rate CONSTANT NUMBER := 0.15;   -- نسبة الضريبة ثابتة
   v_price    NUMBER := 200;
   v_total    NUMBER;
BEGIN
   -- السعر بعد إضافة الضريبة
   v_total := v_price + (v_price * c_tax_rate);
   DBMS_OUTPUT.PUT_LINE('Total = ' || v_total);           -- 230
END;
